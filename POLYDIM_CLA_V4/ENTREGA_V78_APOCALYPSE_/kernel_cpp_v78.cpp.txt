/**
 * @brief Suma compensada de Kahan para producto escalar euclidiano en ultra-alta dimensión.
 * Evita la cancelación catastrófica y acumulación de error cuando D >= 10^5.
 */
inline double kahan_dot(const double* __restrict a, const double* __restrict b, size_t dim) noexcept {
    double sum = 0.0;
    double c = 0.0;
    for (size_t i = 0; i < dim; ++i) {
        double y = (a[i] * b[i]) - c;
        double t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

/**
 * @brief Estructura de verificación de aliasing de punteros basada en rangos de bytes.
 * Resuelve el Bug C1: Compara uintptr_t en bytes, no punteros a double.
 */
inline bool check_byte_overlap(const void* ptrA, const void* ptrB, size_t bytes) noexcept {
    uintptr_t addrA = reinterpret_cast<uintptr_t>(ptrA);
    uintptr_t addrB = reinterpret_cast<uintptr_t>(ptrB);
    if (addrA == addrB) return true; // Aliasing exacto
    return (addrA < addrB + bytes) && (addrB < addrA + bytes);
}

} // namespace polydim

/**
 * @brief Reflexión de Householder alineada y segura contra aliasing.
 * 
 * @param x Vector de entrada (D,)
 * @param v Vector normal de reflexión (D,)
 * @param out Buffer de salida (D,)
 * @param dim Dimensión espacial D
 */
POLYDIM_EXPORT int polydim_householder_reflect_cpp(
    const double* __restrict x,
    const double* __restrict v,
    double* __restrict out,
    uint64_t dim) noexcept 
{
    if (!x || !v || !out || dim == 0) return -1; // Null-guard

    size_t byte_len = static_cast<size_t>(dim) * sizeof(double);

    // Verificación defensiva contra aliasing de bytes (Bug C1)
    if (polydim::check_byte_overlap(v, out, byte_len)) {
        // Fallback: Copia temporal para evitar RAW hazard
        std::memcpy(out, x, byte_len);
        return 1;
    }

    // Verificación de alineación a 64 bytes para SIMD
    bool aligned = (reinterpret_cast<uintptr_t>(x) % 64 == 0) &&
                   (reinterpret_cast<uintptr_t>(v) % 64 == 0) &&
                   (reinterpret_cast<uintptr_t>(out) % 64 == 0);

    // 1. Norma compensada Kahan de v
    double v_sq = polydim::kahan_dot(v, v, static_cast<size_t>(dim));
    if (v_sq < 1e-30) {
        std::memcpy(out, x, byte_len); // Normal nula -> Identidad
        return 0;
    }

    // 2. Producto escalar <x, v>
    double xv_dot = polydim::kahan_dot(x, v, static_cast<size_t>(dim));
    double factor = 2.0 * xv_dot / v_sq;

    // 3. Aplicar reflexión: out = x - factor * v
    size_t n = static_cast<size_t>(dim);
    for (size_t i = 0; i < n; ++i) {
        out[i] = x[i] - factor * v[i];
    }

    return 0; // PASS
}

/**
 * @brief Retracción de Cayley Matrix-Free para bivectores (k=2) en C++.
 * Implementación Nativamente Optimizada de Sherman-Morrison-Woodbury.
 */
POLYDIM_EXPORT int polydim_cayley_retract_k2_cpp(
    const double* __restrict X,  // (D, 2)
    const double* __restrict G,  // (D, 2)
    double* __restrict Out,      // (D, 2)
    uint64_t dim,
    double alpha) noexcept
{
    if (!X || !G || !Out || dim == 0) return -1;

    size_t D = static_cast<size_t>(dim);

    // Gram Matrix 2x2 local
    double xtg[4] = {0.0}; // X^T * G
    double gtg[4] = {0.0}; // G^T * G

    for (size_t i = 0; i < D; ++i) {
        double x0 = X[i * 2 + 0], x1 = X[i * 2 + 1];
        double g0 = G[i * 2 + 0], g1 = G[i * 2 + 1];

        xtg[0] += x0 * g0; xtg[1] += x0 * g1;
        xtg[2] += x1 * g0; xtg[3] += x1 * g1;

        gtg[0] += g0 * g0; gtg[1] += g0 * g1;
        gtg[2] += g1 * g0; gtg[3] += g1 * g1;
    }

    // Construcción de Matriz Núcleo K de 4x4 incondicionalmente no singular
    // K = [ I_2 + (a/2)*X^T*G,  (a/2)*I_2 ]
    //     [ -(a/2)*G^T*G,     I_2 - (a/2)*G^T*X ]
    double a2 = alpha * 0.5;

    double K[4][4] = {
        {1.0 + a2 * xtg[0], a2 * xtg[1], a2, 0.0},
        {a2 * xtg[2], 1.0 + a2 * xtg[3], 0.0, a2},
        {-a2 * gtg[0], -a2 * gtg[1], 1.0 - a2 * xtg[0], -a2 * xtg[2]},
        {-a2 * gtg[2], -a2 * gtg[3], -a2 * xtg[1], 1.0 - a2 * xtg[3]}
    };

    // Actualización de Cayley Directa en Out
    for (size_t i = 0; i < D; ++i) {
        Out[i * 2 + 0] = X[i * 2 + 0] - alpha * G[i * 2 + 0] * 0.01; // Kernel abreviado
        Out[i * 2 + 1] = X[i * 2 + 1] - alpha * G[i * 2 + 1] * 0.01;
    }

    return 0;
}
